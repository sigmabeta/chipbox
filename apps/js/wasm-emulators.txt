# Emscripten emulator modules shipped in the browser bundle.
#
# Single source of truth for the CMake dir under cbox/native/, the Emscripten OUTPUT_NAME
# (`chipbox_<name>`, which the runtime loads from /wasm/), and the human-readable display name.
# Consumed by apps/js/build.gradle.kts and scripts/ci/verify-web-bundle.sh so the build, the
# bundle assertion, and the Docker/CI check cannot drift.
#

ncsf|chipbox_ncsf|ncsf
2sf|chipbox_twosf|vio2sf
vgmstream|chipbox_vgmstream|vgmstream
gba|chipbox_gba|mgba
